# Escort (spec 36, 1850 to 1853): accept, the freighter comes to the players and hails, the answer in its comms starts the route,
# it steers to the jump point, and arrival completes the objective.
# missions: random
newgame run
wait 8
hc_generate escort
hc_accept escort
expect hc_active_escort == 1
# Route (spec 36): MissionTunables.EscortRouteShare (0.65) of the nearest gate distance, so it scales with the system.
expect hc_escort_route_share >= 0.6
expect hc_escort_route_share <= 0.7
# At least 0.65 of the inner gate ring (9,375 at SystemScale 0.5).
expect hc_escort_route >= 6000
expect hc_escort_marker == 1
# Convoy cover: the offer carries MissionTunables.EscortHeatChange, and the reward lowers heat. Heat stays below the first stage, so no hunter disturbs the escort.
expect hc_escort_cover == -8
heat 20
hc_cover_give
expect heat <= 13
# Spec 1851: the players wait 1,800 away. The freighter comes to them and stops close.
hc_escort_goto 1800
# The game can pause by itself here (enemy spotted or a codex page, with default settings in test slots).
pause off
# Spec 1852: it hails when a player hull is within EscortHailGap (800). The comms button flashes.
waitfor hc_escort_hailed == 1 60
expect hc_escort_hailing == 1
expect hc_escort_gap <= 800
expect hc_escort_waiting == 1
expect hc_escort_steering == 0
# "Not yet" only closes the window. "Ready" is the shared choice that starts the route.
hc_escort_call
comms_click Not yet
wait 3
expect hc_escort_waiting == 1
hc_escort_call
comms_click Ready
waitfor hc_escort_waiting == 0 10
wait 3
expect hc_escort_steering == 1
hc_mark
hc_escort_diag before_flight
# The freighter type is random, and a slow one can take a while to turn and speed up. Wait for progress instead of a fixed time.
waitfor hc_escort_progress > 50 45
hc_escort_diag after_flight
expect hc_escort_done == 0
hc_escort_arrive
wait 3
expect hc_escort_done == 1
# Completion removes the destination marker.
expect hc_escort_marker == 0
quit
# tier: smoke
