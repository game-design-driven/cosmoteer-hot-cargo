# Storm survey (spec 136) and sun mining (spec 138).
newgame run
wait 8
ev_set Enabled false
# The test takes several offers of one rival group (spec 103: tow and capture share salvage_rights, sun mining and
# claim jumpers share ore_claims). With rivals on, an open contract dismisses every new offer of its group.
contract_set RivalsEnabled off
godmode on
# The start system has no ion storm, so the test lifts the in-storm rule of the markers.
mech_generate survey nonebula
mech_accept survey
expect mech_survey_markers == 3
log sensor
expect mech_survey_done == 0
mech_survey_next
pause off
wait 20
expect mech_survey_done >= 1
mech_survey_next
wait 20
mech_survey_next
wait 20
expect mech_goal_survey == 1
# Sun mining: ore deposits sit on the seam asteroids. Spec 2384: the ore goes over on the shared delivery page at the station, in trips
# (a small hold carries less than 30): 20, then 10. The job completes at the rules' minimum of 30 of 45, and the pay follows the total.
mech_generate sunmining
mech_accept sunmining
mech_sun_goto
wait 5
expect mech_sun_ore_units >= 30
expect mech_deliver_min == 30
expect mech_deliver_max == 45
mech_deliver_goto
wait 3
expect ev_goods_held == 0
ev_res_add yarden_zamir.helion_ore 20
wait 1
expect ev_goods_held == 1
ev_mark
ev_goods_open
expectevent deliver_goods
pick 1 20
choose 1
wait 1
expect mech_goal_sunmining == 0
expect ev_res_yarden_zamir.helion_ore_gain == -20
ev_res_add yarden_zamir.helion_ore 10
ev_goods_open
pick 1 10
choose 1
wait 1
expect mech_goal_sunmining == 1
expect ev_goods_reward > 0
mech_generate sunmining
mech_accept sunmining
mech_expire sunmining
expect mech_state_sunmining == 2
quit
