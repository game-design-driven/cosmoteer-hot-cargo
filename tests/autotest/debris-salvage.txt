# Salvage and ore drops (spec 1247 to 1249, 2370, 1832, 1834, 1835): a broken junk part drops loose resources × SalvageDropFactor, and a broken ore deposit
# drops its ore × the mining factor (any break) or the laser factor (salvage damage). Exact counts: a laser break of an iron deposit
# drops its 1 iron at 100 % (Fast economy), and a salvage factor of 0 drops nothing.
newgame
wait 8
ev_set Enabled false
debris_set Enabled false
openspace 3000 1500
# Spec 1247: with the factor at 0, broken junk parts drop nothing; at 1, the armor drops steel as in vanilla.
debris_set SalvageDropFactor 0
debris_spawn 10 10
debris_break debris weapon
wait 1
expect debris_broken == 100
expect debris_parts_left == 0
expect debris_loose_gain == 0
debris_set SalvageDropFactor 1
debris_spawn 10 10
debris_break debris weapon
wait 1
expect debris_loose_gain > 0
# Spec 1832: on a weapon break, cheap resources and expensive components get their own cut. A large control room costs steel,
# coil2 and processors. With the base cut at 0, only processors drop; with the component cut at 0, steel drops and no processor does.
debris_set SalvageDropFactor 0
debris_set SalvageComponentDropFactor 1
debris_spawn 5 1 0 cosmoteer.control_room_large
debris_break debris weapon
wait 1
expect debris_gain_steel == 0
expect debris_gain_processor > 0
debris_set SalvageDropFactor 1
debris_set SalvageComponentDropFactor 0
debris_spawn 5 1 0 cosmoteer.control_room_large
debris_break debris weapon
wait 1
expect debris_gain_processor == 0
expect debris_gain_steel > 0
# Spec 1834: a laser break drops 90 % of each part's processors for sure, whatever the cuts: 5 rooms × 10 → 45. Steel and coil keep
# the roll with LaserBaseDropFactor: at 0 no steel drops.
debris_set SalvageComponentDropFactor 0
debris_set LaserBaseDropFactor 0
debris_spawn 5 1 0 cosmoteer.control_room_large
debris_break debris salvage
wait 1
expect debris_gain_processor == 45
expect debris_gain_steel == 0
debris_set LaserBaseDropFactor 0.75
debris_set SalvageComponentDropFactor 0.85
debris_set SalvageDropFactor 0.6
# Spec 1835: ammo in a bay drops none when weapons break the bay, and none when a salvage beam cuts it (LaserBayAmmoShare 0).
# At share 1 the beam drops all of it, as vanilla does.
debris_cargo 40 bullet
wait 1
debris_break debris weapon
wait 1
expect debris_gain_bullet == 0
debris_cargo 40 bullet
wait 1
debris_break debris salvage
wait 1
expect debris_gain_bullet == 0
debris_set LaserBayAmmoShare 1
debris_cargo 40 bullet
wait 1
debris_break debris salvage
wait 1
expect debris_gain_bullet == 40
debris_set LaserBayAmmoShare 0
debris_split
# Spec 2370: cargo comes back in full. A broken storage bay drops all of its steel (vanilla drops half); the bay itself drops nothing at 0.
# Spec 2410: a steel stack is 80, so the 4 tiles of a 2x2 bay hold 320 (vanilla 160).
debris_set SalvageDropFactor 0
debris_cargo 320
wait 1
debris_break debris weapon
wait 1
expect debris_loose_gain == 320
debris_set SalvageDropFactor 1
# Spec 1248: the laser factor scales the ore. At 2, each deposit drops exactly 2 iron; at 1, exactly 1.
debris_set LaserMiningYieldFactor 2
debris_rock 10
debris_break rock salvage
wait 1
expect debris_broken == 10
expect debris_parts_left == 0
expect debris_loose_gain == 20
debris_set LaserMiningYieldFactor 1
debris_rock 10
debris_break rock salvage
wait 1
expect debris_loose_gain == 10
# The default 1.3: between 10 and 20, about 13.
debris_set LaserMiningYieldFactor 1.3
debris_rock 10
debris_break rock salvage
wait 1
expect debris_loose_gain >= 10
expect debris_loose_gain <= 20
quit
