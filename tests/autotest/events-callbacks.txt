# Callbacks (spec 2500 to 2509): an accepted surrender leaves a memory. The next sector brings that ship back by name with its callback
# page. Each memory comes back once, and the big event pacer keeps the next callback for its gap.
newgame run
wait 8
ev_set ArrivalEventChance 0
ev_set StationEventChance 0
ev_set CourierChance 0
ev_set CooldownSeconds 0
ev_callback_chance all 1
ev_set MaxEventsPerSystem 99
godmode on
aiset FleeEnabled false
openspace 6000 3000
# A pirate (a ship with no faction) surrenders to the players' demand: the run remembers it.
dip_set SurrenderChanceBase 5
dip_spawn enemy 250
wait 2
dip demandsurrender enemy
wait 1
expect dip_surrendered == 1
expect ev_memories == 1
# MemoryKind.SparedPirate
expect ev_memory_kind == 2
# Not in the same sector: a memory comes back at least one sector later.
ev_arrive
wait 8
expect ev_recall_kind == 0
# The memory survives a save and a load.
save callbacks
load callbacks
wait 2
expect ev_memories == 1
ev_set ArrivalEventChance 0
ev_set StationEventChance 0
ev_set CourierChance 0
ev_set CooldownSeconds 0
ev_callback_chance all 1
ev_set MaxEventsPerSystem 99
# The next sector: the pirate's ship hails with its callback page, under its old name.
jump
waitfor sector == 2 25
pause off
waitfor ev_recall_kind == 2 20
expect ev_recall_named == 1
expect ev_carrier == 1
choose last
expect ev_pending == 0
# User rule 2026-11-26: each callback kind comes back once per run. Pirate's Honor and Unfinished Business share one kind.
ev_remember SparedPirate
expecteligible cb_pirate_debt 0
expecteligible cb_pirate_grudge 0
# The pacer holds the next callback for its gap (Pursuit.Events, kind Callback).
ev_remember Spared
ev_arrive
wait 8
expect ev_recall_kind == 0
expect ev_pending == 0
# User rule 2026-11-26: up to 4 callbacks per run, each of another kind. Without the gaps, three more kinds come back, and a fifth
# callback does not open.
pu_pace Callback 4 0
pu_set MinEventGapSeconds 0
ev_arrive
waitfor ev_recall_kind == 1 20
choose last
ev_remember PaidOff
ev_arrive
waitfor ev_recall_kind == 3 20
choose last
ev_remember Bluffed
ev_arrive
waitfor ev_recall_kind == 4 20
choose last
ev_remember Rescued
ev_arrive
wait 8
expect ev_pending == 0
expecteligible cb_rescued 1
# Every callback page opens on its memory under the memory's name (a direct open skips the trigger limits).
event cb_rescued
expect ev_recall_kind == 5
expect ev_recall_named == 1
choose 1
expect ev_pending == 0
quit
