# Spec 2520 to 2529: tech that stays in a system when the players leave it is lost. A move that leaves all of it behind loses
# the run at once on arrival (spec 2527), not after the loss timer (600 s here).
newgame
wait 8
ev_set Enabled false
tech_set LossGraceSeconds 600
pause off
jw_twin save
jw_twin spawn
wait 2
expect tech == 8
# Only the twin without tech jumps; the carrier stays with all the tech.
jw_jump_split
waitfor sector == 2 30
expect techleftlost == 8
waitfor lost == 1 10
quit
