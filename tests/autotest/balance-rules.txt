expect balance_jump_cost_per_mass_ly >= 0.0019
expect balance_jump_cost_per_mass_ly <= 0.0021

expect balance_fuel_stack == 10
menu open

# Spec 303: the peak tier counts at most start tier + 2 per sector after the first.
expect balance_peak_tier_cap == 1

# Spec 304: every trade station keeps the fuel floor (tier 1: 10 fuel), the stations near the start too.
expect balance_station_fuel_floor == 10
expect balance_station_fuel_gap >= 0

# still holds the floor (the held fuel stays the same, balance_station_fuel_move checks it).
balance_station_fuel_move
expect balance_station_fuel_gap >= 0
# #307: vanilla crashed when a trade took more fuel from a crew member than it carried.
balance_crew_overdraw

expect balance_fame_heat_gap <= 0.2
