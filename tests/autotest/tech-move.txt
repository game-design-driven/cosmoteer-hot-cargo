# Spec 447: a vault that the player moves with the vanilla grab tool gets rebuilt by crew at the new spot, and holds tech again.
newgame run
wait 8
ev_set Enabled false
tech_set LossGraceSeconds 600
vault steel 64
# An empty vault: crew take it apart and build it again outside the hull.
tech remove 4
wait 1
expect tech == 4
vault grab empty
wait 1
vault place
wait 2
waitfor vaultsbusy == 0 300
tech dump
expect vaults == vaultsneeded
expect vaultsbuilt == vaults
expect shipvalid == 1
tech add 4
wait 1
expect tech == 8
# A vault with tech: taking it apart is salvage, so it costs no tech (spec 65). The tech stays aboard or drops as crates.
vault grab full
wait 1
vault place
wait 2
waitfor vaultsbusy == 0 300
tech dump
expect vaults == vaultsneeded
expect vaultsbuilt == vaults
expect shipvalid == 1
expect techall == 8
# The moved vault holds tech again: the vaults fill to their 8 units (the rest goes to normal storage).
tech add 8
wait 1
expect vaulttech == 8
quit
