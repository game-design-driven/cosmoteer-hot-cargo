# Spec 447: a vault that the player moves with the vanilla grab tool gets rebuilt by crew at the new spot, and holds tech again.
newgame
wait 8
ev_set Enabled false
tech_set LossGraceSeconds 600
vault steel 64
# An empty vault: crew take it apart and build it again outside the hull.
tech remove 4
wait 1
expect tech == 4
vault grab empty
wait 1
vault place
wait 2
waitfor vaultsbusy == 0 300
tech dump
expect vaults == vaultsneeded
expect vaultsbuilt == vaults
expect shipvalid == 1
tech add 4
wait 1
expect tech == 8
# Spec 1554: crew take apart a full vault while the other vault is full (the playtest case). The tech waits on the ship for the
# rebuilt vault: no crate, not in normal storage (it would decay). It counts in the fleet tech and the heat panel, and survives save and load.
vault grab full
wait 1
vault place
waitfor vaultrehomed >= 1 300
expect vaultwaiting == 4
expect tech == 8
expect techcrates == 0
expect storagetech == 0
waitfor panelwaiting == 1 10
save techmove
load techmove
wait 1
expect livetech == 8
# The rebuilt vault takes the waiting tech.
waitfor vaultsbusy == 0 300
waitfor vaultwaiting == 0 60
tech dump
expect vaults == vaultsneeded
expect vaultsbuilt == vaults
expect shipvalid == 1
expect vaulttech == 8
expect tech == 8
expect techcrates == 0
expect storagetech == 0
quit
