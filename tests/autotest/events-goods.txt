# Spec 2384: a vanilla Resource Delivery offer is a goods hand-over at its station: the trade tab does not deliver, the shared page does.
# The amount is fixed (vanilla picks it), so the page shows one option; handing it over pays the full reward at the station.
# missions: random
newgame run
wait 8
ev_set Enabled false
ev_set StationEventChance 0
contract_set RivalsEnabled off
contract_pick any ResourceDelivery
contract_terms fee 0
contract_terms informant 0
contract_accept
expect contract_accepted == 1
expect ev_goods_held == 0
ev_goods_supply
expect ev_goods_held == 1
ev_contact_goto
pause off
wait 3
ev_goods_open
expectevent deliver_goods
ev_goods_pick
choose 1
wait 1
expect ev_handover == 1
expect ev_goods_reward > 0
quit
