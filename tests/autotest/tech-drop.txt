# Destroyed vaults (spec 65, 445): crates or burned tech by a forced roll. A new vault that crew build takes the crates back (spec 447).
newgame run
wait 8
ev_set Enabled false
tech_set LossGraceSeconds 600
# A new empty vault, built like a player builds it (the career construction mode: crew and steel).
vault steel 32
vault build
waitfor vaultsbuilt > vaultsneeded 240
expect outsidetech == 0
# Roll forced to drop: a destroyed vault leaves its 4 units as crates.
tech_set VaultTechLossChance 0
vault destroy
wait 2
expect tech == 4
expect techcrates == 4
expect vaultburned == 0
# Roll forced to burn: the tech is gone, and a message says so. No tech is aboard, so the loss timer runs.
tech_set VaultTechLossChance 1
vault destroy
wait 2
expect tech == 0
expect techcrates == 4
expect vaultburned == 1
expect losstimer == 1
# Crew bring the crates into the new vault, the only one with room. The timer stops.
tech collect
waitfor losstimer == 0 90
waitfor outsidetech == 4 240
expect storagetech == 0
expect lost == 0
quit
