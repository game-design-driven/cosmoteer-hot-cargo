# Escort guard (spec 1881 to 1886): after "Ready" the freighter moves only while a player ship is near. Alone it holds, a red ring
# follows it, the alone line counts down, raiders find it after EscortRaidSeconds, and raiders take it when its time alone runs out.
# Before this rule the freighter flew alone past raiders that the game had frozen in stasis, and arrived.
# missions: random
newgame
wait 8
ev_set Enabled false
godmode on
hc_generate escort
hc_accept escort
hc_escort_goto
pause off
wait 3
hc_escort_ready
wait 2
expect hc_guard_alone == 0
# Spec 1886: Convoy Duty's rules give it 4 minutes alone.
expect hc_guard_lost_after == 240
# Spec 1881: the players 1,800 away (more than EscortGuardRadius, less than the 2,500 live range), so the freighter is live, but it holds.
hc_escort_goto 1800
pause off
waitfor hc_guard_alone == 1 10
expect hc_guard_live == 1
# Spec 1886: the red ring follows it, and the line counts down from 4:00.
expect hc_guard_ring == 1
expect hc_guard_left <= 240
expect hc_guard_left >= 225
hc_mark
wait 10
expect hc_escort_progress < 300
# Back next to it: it flies on, and the time alone starts again from 0.
hc_escort_goto
pause off
waitfor hc_escort_progress > 400 45
expect hc_guard_alone == 0
expect hc_guard_exposed == 0
expect hc_guard_ring == 0
# Spec 1882: alone for EscortRaidSeconds (60 s, here set to 59 s), live raiders jump in around it and attack it.
hc_escort_goto 1800
pause off
waitfor hc_guard_alone == 1 10
hc_guard_expose 59
waitfor hc_guard_raiders >= 2 10
waitfor hc_guard_raiders_on_escort >= 1 30
expect hc_guard_failed == 0
# Spec 1883, 1886: far away (it goes into stasis), the clock still runs. When it runs out, raiders take it: it turns hostile, and
# the failed mission leaves the log.
hc_guard_clear_raiders
hc_guard_mark
hc_escort_goto 9000
pause off
waitfor hc_guard_live == 0 20
expect hc_guard_failed == 0
# The second warning, one minute before the end.
hc_guard_expose 179
wait 2
expect hc_guard_left <= 60
hc_guard_expose 239
wait 2
expect hc_active_escort == 0
expect hc_guard_marked_taken == 1
quit
