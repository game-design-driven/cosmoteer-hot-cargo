# The troublemaker (spec 480 to 485): a ship pays the players to take a troublemaker aboard. A thief steals twice the pay at the next
# station visit and slips off; an honest one stays. One branch each, with the roll fixed by the chance (0 or 1).
newgame run
wait 8
ev_set StationEventChance 0
ev_set ArrivalEventChance 0
ev_set CourierChance 0
ev_set CooldownSeconds 0
godmode on
pause off
ev_tostation 2000
# Spec 484: a refusal. The ship leaves; no pay, no crew.
event troublemaker
ev_carrier_goto
fr_money_mark
ev_comms carrier Not our problem
expect ev_pending == 0
expect fr_trouble == 0
expect fr_money_delta == 0
expect fr_crew_delta == 0
# The starter fleet has no free bunk; the accept needs one.
fr_free_bunk
# Spec 481 to 483: a thief. The pay and one crew member on accept; at the station they take twice the pay and leave.
fr_set TroublemakerThiefChance 1
event troublemaker
ev_carrier_goto
fr_money_mark
ev_comms carrier Take them aboard
expect fr_trouble == 1
expect fr_trouble_thief == 1
expect fr_money_delta == fr_trouble_pay
expect fr_crew_delta == 1
wait 3
expect fr_trouble == 1
fr_money_mark
ev_tostation 100
waitfor fr_trouble == 0 10
expect fr_money_delta == fr_trouble_theft
expect fr_crew_delta == -1
# Spec 482: an honest one stays as normal crew after the station visit.
ev_tostation 2000
fr_set TroublemakerThiefChance 0
event troublemaker
ev_carrier_goto
ev_comms carrier Take them aboard
expect fr_trouble_thief == 0
fr_money_mark
ev_tostation 100
waitfor fr_trouble == 0 10
expect fr_money_delta == 0
expect fr_crew_delta == 0
# Spec 484: a thief who dies before the station visit steals nothing. The honest one has the free bunk now.
ev_tostation 2000
fr_free_bunk
fr_set TroublemakerThiefChance 1
event troublemaker
ev_carrier_goto
ev_comms carrier Take them aboard
expect fr_trouble == 1
fr_trouble_kill
waitfor fr_trouble == 0 5
fr_money_mark
ev_tostation 100
wait 3
expect fr_money_delta == 0
quit
