# Friendly ships (spec 574 to 583): hold tech and its return, the fee, betrayal, held tech against the loss rule,
# hired help that joins and runs, the wanted rule, mercenaries that hail and take pay by the minute.
newgame run
wait 8
ev_set Enabled false
aiset FleeEnabled false
aiset CallEnabled false
aiset RegroupEnabled false
# Personalities roll with the start world; a cover or bait personality would keep the hired ship from attacking.
aiset PersonalitiesEnabled false
dip_set DemandHailChance 0
dip_set MercChance 0
expect dip_vaulttech == 8
# A friendly warship answers calls and shows the topics.
dip_spawn friend 250
wait 2
expect dip_answers == 1
dip_page friend
expect dip_pagebuttons >= 2
# Hold and help each have their own slider in their row (spec 591).
expect dip_pageslider == 2
# Spec 2223: asking to hold tech is the last option row.
expect dip_holdlast == 1
# Hold for free, then the return without betrayal. Held tech counts in the fleet tech.
dip_set HoldChanceBase 5
dip_set FeeChanceBase -5
dip_set BetrayChanceBase -5
dip hold friend 3
wait 1
expect dip_held == 3
expect dip_vaulttech == 5
expect tech == 8
dip return friend
wait 1
expect dip_held == 0
expect dip_vaulttech == 8
# Hold is once per ship.
dip hold friend 2
wait 1
expect dip_held == 0
# A second ship asks a fee first; paying it starts the hold. Then it betrays at the return.
dip_spawn friend 250
wait 2
dip_set FeeChanceBase 5
dip_set FeePerUnit 0.01
dip hold friend 2
wait 1
expect dip_asks == 1
expect dip_held == 0
dip_mark
dip payholdfee friend
wait 1
expect dip_asks == 0
expect dip_held == 2
expect dip_moneyspent > 0
dip_set BetrayChanceBase 5
dip return friend
wait 1
expect dip_held == 0
expect dip_vaulttech == 6
expect tech == 6
# Held tech keeps the run going: a third ship holds 2, then the players lose all vault tech.
dip_spawn friend 250
wait 2
dip_set FeeChanceBase -5
dip hold friend 2
wait 1
expect dip_held == 2
tech remove 4
wait 1
expect dip_vaulttech == 0
wait 6
expect tech == 2
expect lost == 0
dip_log held
# Hired help: a fourth ship joins for pay, attacks an enemy near the player and runs when badly hurt.
godmode on
dip_spawn friend 300
wait 2
# The first paid hire at the sure price (spec 579) always works with no heat and reputation factors.
dip_set HirePerHeat 0
dip_set HirePerReputation 0
dip_page friend
expect dip_pageslider >= 1
expect dip_trust == 1
dip_expect hire friend 10 == 1
dip_expect hire friend 6 == 0.6
dip hire friend 10
wait 1
expect dip_hires == 1
expect dip_paidhires == 1
dip_spawn enemy 400
# The hired ship attacks within a few AI checks. In CI it had not after 4 s (#184), so the step waits, and the report logs the hire's
# command and the ships near the player for the next failure.
wait 2
dip_hire_report
waitfor dip_friendattacking == 1 30
dip_damage friend 0.3
waitfor dip_hirefleeing == 1 10
dip_log fleeing
# Spec 2224: the hit counts against later hires: trust = 1 / (1 + 1 x 0.15) / (1 + lost x 0.5).
wait 2
expect dip_shipslost >= 0.2
expect dip_trust < 0.8
# Mercenaries: an armed friendly ship near a station rolls mercenary, hails, and takes pay by the minute (slider).
tostation
wait 2
dip_set MercChance 1
dip_set MercHailChance 1
dip_spawn friend 250
wait 4
expect dip_friendmerc == 1
expect dip_hailing >= 1
dip_page friend
expect dip_pageslider >= 1
dip_mark
dip merc friend 2
wait 1
expect dip_friendhired == 1
expect dip_moneyspent > 0
expect dip_merc_cheaper == 1
expect dip_paidhires == 2
dip_log merc
# Spec 2224: a ship is never hired twice. After its contract ends, the mercenary takes no new one.
dip_endhire friend
wait 2
expect dip_friendhired == 0
dip merc friend 2
wait 1
expect dip_friendhired == 0
expect dip_paidhires == 2
# A military ship of the corporation's ally faction never holds tech (spec 574). A friendly military ship answers only while it has
# a real option: after its help offer is used, it keeps the vanilla behavior (spec 560).
dip_set MercChance 0
dip_set HoldChanceBase 5
# Outside corporate ally space: there the ally faction takes no hire contract (spec 1150, test factions-peace).
fa_roles other -
dip_spawn ally 300
wait 2
expect dip_answers == 1
dip_page friend
expect dip_pageslider == 1
dip hold friend 2
wait 1
expect dip_held == 2
# Almost no trust left: the lowest offer fails, and the money stays.
dip_set HireUseFalloff 1000
dip hire friend 6
wait 1
expect dip_friendhired == 0
expect dip_answers == 0
# Wanted: nobody helps.
wanted
wait 1
dip_set MercChance 0
dip_spawn friend 300
wait 2
dip_set HireUseFalloff 0.15
dip_expect hire friend 10 == 0
dip hire friend 10
wait 1
expect dip_friendhired == 0
quit
