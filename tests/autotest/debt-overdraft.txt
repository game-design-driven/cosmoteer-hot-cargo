# Money below 0 (spec 1417 to 1419). A spend that takes money below 0 opens a debt with the system's faction. Its lender gets
# in touch 60 s later above 20,000, once; not at 20,000 or less; not after the debt is paid. The money tooltip lists each lender.
# The overdraft reads the money value in each tick, so any spend counts. Limit: a single vanilla build in the sim checks the money
# before it buys, so debt_spend stands in for the build that went below 0 in the playtest.
newgame run
wait 8
ev_set Enabled false
godmode on
debt_set LoanStationChance 0
pu_setmoney 100
debt_spend 900
waitfor debt_mine == 1 10
expect debt_my_money >= 0
expect debt_lender_owner == 1
expect debt_hail_due == 1
# Above 20,000: the lender's collector comes 60 s after the debt started, once, and its page is the debt page.
debt_my_set 25000
wait 30
expect debt_overdraft_collectors == 0
waitfor debt_overdraft_collectors == 1 40
expect debt_hail_due == 0
waitfor debt_hailed == 1 60
debt_look collector
expect debt_page_debt_parts >= 1
expect debt_page_other_lines == 0
debt_notnow
wait 2
# At 20,000 or less: no hail.
debt_my_clear
wait 1
debt_spend 500
waitfor debt_mine == 1 10
debt_my_set 20000
wait 70
expect debt_hail_due == 0
expect debt_overdraft_collectors == 0
# Paid before the hail: no hail.
debt_my_clear
wait 1
debt_spend 500
waitfor debt_mine == 1 10
debt_my_set 30000
wait 20
debt_my_clear
wait 60
expect debt_overdraft_collectors == 0
# Spec 1419: the money tooltip lists each lender with its balance.
debt_spend 500
waitfor debt_mine == 1 10
debt_owe other 5000
wait 1
expect debt_money_tip_lenders == 2
quit
