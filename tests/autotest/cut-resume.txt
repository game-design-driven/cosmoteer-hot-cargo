# Cut ships apart (spec 2464): when crew leave the cut jobs, the edges keep their work, the jobs go back to the queue, and the cut
# ends after the crew come back.
newgame
wait 8
ev_set StationEventChance 0
ev_set ArrivalEventChance 0
ev_set CourierChance 0
godmode on
pause off
wait 3
cut_mark
cut_clean
waitfor cut_work > 1 90
cut_interrupt
expect cut_crew == 0
expect cut_work > 1
expect cut_splits == 0
waitfor cut_crew > 0 60
waitfor cut_splits == 1 240
expect cut_orders == 0
quit
