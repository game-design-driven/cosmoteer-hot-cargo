# Tow (spec 134) and capture (spec 135, 2380: the bank takes the freighter at the station).
# missions: random
newgame run
wait 8
ev_set Enabled false
# The test takes several offers of one rival group (spec 103: tow and capture share salvage_rights, sun mining and
# claim jumpers share ore_claims). With rivals on, an open contract dismisses every new offer of its group.
contract_set RivalsEnabled off
godmode on
mech_generate tow
mech_accept tow
mech_tow_goto
pause off
wait 4
expect mech_tow_junk == 1
expect mech_goal_tow == 0
mech_tow_deliver
wait 3
expect mech_goal_tow == 1
# Tow failure: the hulk is destroyed.
mech_generate tow
mech_accept tow
mech_tow_goto
wait 4
mech_tow_remove
wait 2
expect mech_state_tow == 2
# Tow failure: the time limit runs out.
mech_generate tow
mech_accept tow
mech_expire tow
expect mech_state_tow == 2
# Capture by surrender.
mech_generate capture
mech_accept capture
mech_capture_goto
wait 4
expect mech_capture_hostile == 1
mech_capture_surrender
wait 2
expect mech_goal_capture == 1
# The surrendered freighter stays in the system (spec 2380). Alongside the bank's station it is delivered at once, with no comms step
# (spec 1877: the shared arrival rule of ship deliveries).
wait 10
expect ev_handover == 0
mech_capture_goto
pause off
wait 4
ev_deliver_ship
pause off
wait 3
expect ev_handover == 1
# Capture by crew: the freighter becomes a wreck, then the players' crew claim it.
mech_generate capture
mech_accept capture
mech_capture_goto
wait 4
mech_capture_claim
wait 2
expect mech_goal_capture == 1
# Player crew aboard: the bank waits until they are off, and the objective line says so. The claimed freighter is a player ship, so it
# never goes into stasis.
ev_delivered_crew 1
ev_contact_goto
pause off
wait 3
ev_deliver_ship
pause off
wait 3
expect ev_handover == 0
expect ev_ship_wait == 1
ev_delivered_crew 0
wait 2
expect ev_handover == 1
# Capture failure: the freighter is destroyed.
mech_generate capture
mech_accept capture
mech_capture_goto
wait 4
mech_capture_remove
wait 2
expect mech_state_capture == 2
quit
