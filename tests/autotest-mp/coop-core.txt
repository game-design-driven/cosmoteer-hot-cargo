# Co-op session: host tunables, heat and hunters, settings menu, run modes, ship tactics, resync, save and load.
# The session file is the host's script. "client:" lines run on the client before it joins; the client then follows (client.txt).
# "as 1 <command>": the command runs in the sim as the client player. "on 1 <command>": it runs on the client's own machine.
# Every step logs its own result. Both machines log a state hash every 300 ticks; scripts/coop-report.sh compares them.

# Before the join. The host's tunables.rules differs from the client's: both start with the defaults, then the host's file
# arrives as an MP input.
tunablesfile Heat { HeatPerGameMinute = 30; HeatCarryOverFraction = 0.5 }  Events { Enabled = false }
mphost {PORT}
mpstart {PLAYERS}
mp godmode on
# For the whole session the player ships take no damage and lose no crew: no step tests who wins a fight. CI 2026-11-17: random
# fights took the client's only ship. Queue run 45239133782: the fights before shipai.fight killed the client's crew, and a ship
# without crew takes no crew orders. Queue run 45513513863 (#305): with the shield off, the very hard wave of runmodes.difficulty
# destroyed the client's only ship. The shield is a flag on each machine, so it stays on through the resync and the load. A step
# that needs damage turns it off and on again itself.
mp shieldplayers on
mp maxhunters 4
mp tacticlog on
# Small telemetry relay chunks on the client, so the session sends many of them (step cut.split
# Spec 2469: the client cuts its own ship. The order, the crew work and the split run in the sim on both machines.
on 1 cut_mark
on 1 cut_clean
waitfor cut_orders == 1 30
on 1 waitfor cut_orders == 1 30
waitfor cut_splits == 1 240
on 1 waitfor cut_splits == 1 240

step telemetry.relay).
on 1 telerelaychunk 4096

step core.tunables
waitfor heatperminute == 30 30
on 1 waitfor heatperminute == 30 30

step core.client-heat
# The client adds heat, and the first stage (Warning at heat 40, #19) starts on both machines.
as 1 heat 40
waitfor stages >= 1 60

step ux.ping
# Spec 1500 to 1506: the client pings its own ship (Alt + left click). Both machines show one ping, as a vanilla indicator, on
# the ship's center (it follows the ship). The client's next ping, in space, replaces it. A short lifetime ends it on both.
on 1 ping own
waitfor ping_count == 1 30
on 1 waitfor ping_count == 1 30
waitfor ping_on_ship == 1 10
on 1 waitfor ping_on_ship == 1 10
waitfor ping_indicators == 1 10
on 1 ping space 3000 0
waitfor ping_on_ship == 0 30
on 1 waitfor ping_on_ship == 0 30
expect ping_count == 1
mp ux PingSeconds 2
waitfor ping_count == 0 30
on 1 waitfor ping_count == 0 30
waitfor ping_indicators == 0 10
mp ux PingSeconds 30

step core.hunters
# Hunters jump in, the host forces two more stages, and the hunters stay hostile to both players.
waitfor hunters >= 1 60
mp stage
mp stage
waitfor stages >= 3 30
waitfor hunters >= 2 60
wait 20
hunters host
expect huntershostile == 1
on 1 hunters client
on 1 expect huntershostile == 1

step hunters.capture
# Playtest 2026-11-15: the players took over a hunter, repaired it and jumped it to a station, then the game desynced.
# Spec 2280: a taken hunter is a normal player ship on both machines and stays. Spec 1510: money that the parallel ship update
# adds (a refund of expired pre-purchases) is never lost, so the money of both players is the same on both machines.
mp takehunters
wait 2
expect hunterstaken >= 1
expect huntertagplayers == 0
on 1 expect hunterstaken >= 1
on 1 expect huntertagplayers == 0
mp moneyrace 2000
wait 2

step tunablesmenu.host-edit
# Spec 504: the host's menu change reaches the client through the tunables input. A menu edit keeps the file's other values.
waitfor menu_ready == 1 30
menu open
expect menu_readonly == 0
menu set Events.Enabled off
menu set Heat.HeatPerGameMinute 45
menu set Hunters.Stages[0].ShipCount 4
menu close
waitfor heatperminute == 45 30
menu file has HeatPerGameMinute = 45
menu file has HeatCarryOverFraction = 0.5
menu value Heat.HeatCarryOverFraction == 0.5

step tunablesmenu.client-readonly
on 1 waitfor heatperminute == 45 30
on 1 menu open
on 1 expect menu_readonly == 1
on 1 menu reject Heat.HeatPerGameMinute 10
on 1 menu close
on 1 menu value Hunters.Stages[0].ShipCount == 4
on 1 menu value Heat.HeatCarryOverFraction == 0.5

step tunablesmenu.file
# Spec 60, 850: a change of the host's tunables.rules mid-run goes out as the tunables input, and every machine applies it
# at the same tick. A bad file sends nothing. The client's own file has no effect.
tunablesfile Events { Enabled = false }; Heat { HeatPerGameMinute = 50 }; Hunters { MaxLiveHunters = 7 }
waitfor heatperminute == 50 30
menu open
menu shows Heat.HeatPerGameMinute 50
menu close
tunablesfile Heat { HeatPerGameMinute = 55; HeatCarryOverFraction = 2 }
wait 3
expect tunables_error == 1
expect heatperminute == 50
on 1 waitfor heatperminute == 50 30
on 1 menu open
on 1 expect menu_readonly == 1
on 1 menu shows Heat.HeatPerGameMinute 50
on 1 menu close
on 1 menu value Hunters.MaxLiveHunters == 7
on 1 tunablesfile Heat { HeatPerGameMinute = 99 }
on 1 wait 3
on 1 expect heatperminute == 50
wait 2
expect heatperminute == 50

step shipai.tactics
# Spec 409, 411: edits by either player go through the tactics input. The client's edits come from its own machine.
mp aiset FleeEnabled false
# The host's start ship. Ship IDs are random, so "ship 0" can be a hunter of hunters.capture, which has no command points
# and never runs its tactics (vanilla CanExecuteCommands).
tactics_ship Model-L
tactics Defensive
tacticsauto on
on 1 tactics Sniper
on 1 tacticsauto on
waitfor tactics_any_auto >= 2 30
expecttacticsany Sniper
expecttacticsany Defensive

step shipai.fight
# Both machines run the same tactics decisions in a fight.
# Two ships of the sector tier: the Warning response of the last sector (#19 removed the Patrol stage).
mp stageset Warning EarlyScale 1
mp spawnstage Warning
wait 30
# Two start ships with tactics, one evaluation each second.
expect tactics_evals > 20

step core.resync
# The resync that follows a desync (and a late join): every machine reloads the host's game. A pending event, a sprung trap,
# the stage and the ship tactics come back on both machines.
mp hc_generate shady
mp hc_accept shady
mp hc_shady_spring
mp event smuggler_den
waitfor hc_trap_state == 2 20
# The event starts in the sim some ticks after the line, later on a slow machine: wait for it.
waitevent smuggler_den 20
# The fight can destroy a ship's control room (godmode cannot stop a one-hit kill); a ship without command becomes a wreck.
# So the tactics check compares every ship that is still a human ship with a snapshot from just before the resync.
mp tactics_snapshot
wait 1
save mpsave
mpresync
waitfor mpgames >= 2 120
waitfor active == 1 60
expectevent smuggler_den
expect hc_trap_state == 2
expect stages >= 1
expect heat >= 30
expect tactics_snapshot_kept >= 1
expect tactics_snapshot_changed == 0
on 1 waitfor active == 1 60
on 1 expectevent smuggler_den
on 1 expect hc_trap_state == 2
on 1 expect tactics_snapshot_kept >= 1
on 1 expect tactics_snapshot_changed == 0
# The sprung trap's ambushers stay in the system. They held up the escort freighter of the next step (CI 2026-11-17), so they go
# now, and nothing else: a kill of every enemy also took the freighter and stations of a faction that the players were wanted by
# (queue run 45231373438).
mp hc_kill_ambush

step missions.escort-ambush
# Playtest 2026-11-15: after a resync, the convoy freighter reached the ambush point at a different tick on each machine,
# the raiders spawned at different spots, and the game desynced (spec 1512). Objectives read the sim position of their ships.
# No hunter fights the freighter on its way: on main (CI 2026-11-16) a hunter of the steps before held it up past the limit.
mp godmode on
mp maxhunters 0
mp stasishunters
mp hc_generate escort
mp hc_accept escort
mp hc_escort_goto
pause off
wait 3
# Spec 1852: the freighter hails, and the client answers "Ready" in its own comms window (the shared choice input).
# Both machines see the freighter leave.
waitfor hc_escort_hailed == 1 60
# The freighter came to the host's nearest ship. The client brings its own ship into comms reach first.
as 1 hc_escort_bring
wait 2
on 1 hc_escort_call
on 1 comms_click Ready
waitfor hc_escort_waiting == 0 15
on 1 waitfor hc_escort_waiting == 0 15
# Playtest 2026-11-24: after the client's "Ready" the freighter flies the route by itself, before any teleport. The game must run for
# that: queue run 2026-11-24 stopped right after the client's comms window opened (PauseLog now names any pause and who made it).
pause off
waitfor paused == 0 10
hc_mark
waitfor hc_escort_progress > 50 45
# Just before the ambush point (EscortAmbushAt 0.3). The freighter flies on and springs the ambush itself. From 0.25 the
# flight took 11 to 60+ s in CI queue runs (the hunters of the steps before fight it), too close to the 60 s limit.
mp hc_escort_advance 0.29
# The ambush raiders spawn when the freighter reaches 0.3.
pause off
# #308: first the freighter moves on from its new spot. A freighter that does not move fails here, apart from a slow flight (CI: 3 to
# 36 s to the ambush in 60 runs, once never). The mark waits one second, so it is taken after the advance ran in the sim.
wait 1
hc_mark
waitfor hc_escort_progress > 20 30
waitfor hc_escort_raiders >= 2 60
on 1 waitfor hc_escort_raiders >= 2 60
# Spec 1881, 1882: both players leave the freighter. It holds alone, and its raid spawns at the same tick and place on both machines.
mp hc_escort_goto 1800
pause off
waitfor hc_guard_alone == 1 15
on 1 waitfor hc_guard_alone == 1 15
mp hc_guard_expose 59
waitfor hc_guard_raiders >= 2 30
on 1 waitfor hc_guard_raiders >= 2 30
# Spec 1890 to 1892: the Blind Courier answers the client and follows the players on both machines.
mp hc_guard_clear_raiders
mp hc_generate lead
mp hc_accept lead
mp hc_guard_goto
pause off
waitfor hc_lead_hailed == 1 60
as 1 hc_lead_ready
waitfor hc_lead_waiting == 0 15
on 1 waitfor hc_lead_waiting == 0 15
mp hc_lead_lead 700
pause off
# #308: the gap check waits until the move above ran in the sim. Before, the host measured the gap before the players moved, and only
# the client's check was real.
wait 1
waitfor hc_lead_gap < 450 45
on 1 waitfor hc_lead_gap < 450 45
wait 5

step missions.mayday
# Spec 2434: the mayday starts in the sim on both machines, with the same victim and attackers, and ends the same way.
mp hc_mayday_start
waitfor hc_mayday_active == 1 15
on 1 waitfor hc_mayday_active == 1 15
waitfor hc_mayday_attackers >= 2 15
on 1 waitfor hc_mayday_attackers >= 2 15
mp hc_mayday_kill_attackers
waitfor hc_mayday_done == 1 15
on 1 waitfor hc_mayday_done == 1 15

step core.save-load
# Back to the game setup with the connection kept, load the save (from before the resync) for both players, start.
mplobby
mpload mpsave
mpstart {PLAYERS}
# A loaded co-op game starts paused.
pause off
waitfor paused == 0 10
# The save is from the debris steps, so it holds their hunter cap of 0. The cap of tunablesmenu.file again.
mp maxhunters 7
expectevent smuggler_den
expect hc_trap_state == 2
expect stages >= 1
expect heat >= 30
on 1 waitfor active == 1 60
on 1 expectevent smuggler_den
on 1 expect hc_trap_state == 2

step worlds.close-pass
# Spec 2445: Close the Pass in co-op. The clock, the convoy spawn and the win run in the sim on both machines. Built in this system
# (worlds closepass_here), because the session does not jump to a Mining Belt.
mp worlds closepass_here
mp worlds goto closepass
waitfor worlds_closepass_running == 1 15
on 1 waitfor worlds_closepass_running == 1 15
waitfor worlds_closepass_freighters >= 2 15
on 1 waitfor worlds_closepass_freighters >= 2 15
mp worlds kill hc_closepass_freighter
waitfor worlds_closepass_done == 1 15
on 1 waitfor worlds_closepass_done == 1 15
mp worlds kill hc_closepass_escort

step runmodes.difficulty
# Spec 680 to 685: the host changes the difficulty through the vanilla settings inputs. The client sees the same effective
# heat rate, hunter count and tier, tech value and fees. The state hash compares the heat that time adds at the new rate.
rm_difficulty very_hard slow scarce
rm_log host
on 1 expect rm_heatrate > 1
on 1 expect rm_ships > 1
on 1 expect rm_tech < 1
on 1 expect rm_techgap < 0.01
on 1 expect pricegap == 0
on 1 rm_log client
# Two ships of the sector tier: the Warning response of the last sector (#19 removed the Patrol stage).
mp stageset Warning EarlyScale 1
mp spawnstage Warning
wait 15
rm_log host-after
on 1 expect rm_tiers >= 1
on 1 rm_log client-after

step telemetry.relay
# Spec 1592, 1593: the client's telemetry files reach the host's folder in many small chunks: the sessions of the join, the resync
# and the load. Closed files arrive whole; the open file trails by at most the lines of the last moments. No copy breaks the format.
on 1 teleflush
on 1 waitfor tele_relay_pending_closed == 0 60
on 1 waitfor tele_relay_pending < 16384 30
expect tele_relay_copies >= 3
expect tele_relay_bad == 0
expect tele_relay_received > 0


step telemetry.note
# Spec 1986, 1988: the client's playtest note reaches the host's live notes file at once, and the game keeps running.
# Spec 1995, 1996: its small screenshot reaches the host's notes folder, the host's line names that copy, and the host confirms the line.
on 1 telenote shot
on 1 telenote open
on 1 telenote type the client saw this
on 1 telenote save
waitfor telenote_live_others >= 1 30
on 1 waitfor telenote_open == 0 10
on 1 waitfor telenote_unsent == 0 30
on 1 waitfor tele_relay_pending_shots == 0 60
waitfor telenote_host_shots >= 1 30

step telemetry.crash
# Spec 1997: a client crash ends its session with a crash event, and the rest of its files reaches the host.
on 1 telecrash
waitfor tele_relay_crashes >= 1 60
end
